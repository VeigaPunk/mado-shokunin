def git(repo, *args):

def make_clone(repo, rev, work):
    clone = os.path.join(work, "clone-%d" % os.getpid())
    if os.path.exists(clone):
        shutil.rmtree(clone)
    try:
        git(work, "clone", "-q", "--shared", "--no-checkout", repo, clone)
    except SetupError as exc:
        raise SetupError("clone of %s failed: %s" % (repo, exc))
    try:
        git(clone, "-c", "advice.detachedHead=false", "checkout", "-q", "--detach", rev)
    except SetupError:

    if os.path.realpath(top) != os.path.realpath(clone):
        raise SetupError("clone toplevel %s != %s" % (top, clone))

def tree(clone, rev):
    """mode -> path mapping plus the path list, one `git ls-tree -r` pass."""
    out = git(clone, "ls-tree", "-r", "-z", rev)
    modes, paths = {}, []
    for entry in out.split(b"\x00"):
        if not entry:
            continue
        meta, _, path = entry.partition(b"\t")
        parts = meta.split()
        if len(parts) < 3:
            continue
        mode, oid = parts[0].decode(), parts[2].decode()
        name = path.decode("utf-8", "surrogateescape")
        modes[name] = (mode, oid)
        paths.append(name)
    return modes, paths

def show(clone, rev, rel):

def parse_rows(text):
    rows, fenced = [], False
    for raw in text.splitlines():
        line = raw.strip()
        if line.startswith(("```", "~~~")):
            fenced = not fenced
            continue
        if fenced or not line.startswith("|"):
            continue
        cells = [c.strip() for c in line.strip("|").split("|")]
        first = cells[0].strip(EMPHASIS) if cells else ""
        if not first or "/" not in first or first.startswith(":") or set(first) <= set("-: "):
            continue
        rows.append({
            "path": first,
            "desc": cells[1] if len(cells) > 1 else "",
            "pointer": " | ".join(cells[2:]) if len(cells) > 2 else "",
        })
    return rows


def pointer_tokens(cell):
    tokens = re.findall(r"`([^`\n]+)`", cell)
    if not tokens:
        tokens = re.split(r"[,;]|\s+", cell)
    out = []
    for token in tokens:
        token = token.strip(EMPHASIS)
        token = re.sub(r"^\./", "", token)
        token = LINE_ANCHOR.sub("", token)
        if not token or token.startswith(("/", "~", "$")):
            continue
        if any(ch in token for ch in "<>():#?*\"'"):
            continue
        if "/" not in token and not re.search(r"\.(md|py|sh|json|txt|ya?ml)$", token):
            continue
        out.append(token)
    return out


def norm_desc(cell):
    return re.sub(r"[^a-z0-9]+", " ", cell.strip(EMPHASIS).lower()).strip()


def desc_words(cell):
    return {w.group(0).lower() for w in WORD.finditer(cell)}


def symlink_target(clone, rev, path, depth=0):
    """Resolve a symlink blob at <rev> to a repo path, or None."""
    if depth > 8:
        return None
    try:
        target = show(clone, rev, path).decode("utf-8", "replace").strip()
    except SetupError:
        return None
    if not target:
        return None
    nxt = os.path.normpath(os.path.join(os.path.dirname(path), target)).replace(os.sep, "/")
    return nxt


def ledger_shaped(rows, modes):
    """True when a blob is itself a ledger: enough rows whose pointer cell names repo paths.

    A byte-copy of the ledger under another name still answers every basename, so it is a
    self-pointer by proxy. PROVINCE.md carries an instrument table, but its third cell is a
    prose write-scope column with no repo paths, so it is not ledger-shaped.
    """
    if len(rows) < 4:
        return False
    with_paths = 0
    for row in rows:
        if any(t in modes for t in pointer_tokens(row["pointer"])):
            with_paths += 1
    return with_paths >= max(3, len(rows) // 2)


def pointer_state(clone, rev, modes, row_path, pointer, pop_set):
    """Returns (resolves, self_pointer)."""
    base = os.path.basename(row_path)
    ledger_oid = modes.get(LEDGER_REL, (None, None))[1]
    self_hit = False
    resolves = False
    for token in pointer_tokens(pointer):
        if token not in modes:
            continue
        mode, oid = modes[token]
        try:
            blob = show(clone, rev, token)
        except SetupError:
            continue
        if base.encode() not in blob:
            continue
        resolves = True
        if token in (row_path, LEDGER_REL):
            self_hit = True
            break
        if ledger_oid and oid == ledger_oid:
            self_hit = True
            break
        if mode == "120000":
            target = symlink_target(clone, rev, token)
            if target in (LEDGER_REL, row_path):
                self_hit = True
                break
        table = parse_rows(blob.decode("utf-8", "replace"))
        if table and ledger_shaped(table, modes) and any(
                row["path"] in pop_set for row in table):
            self_hit = True
            break
    return resolves, self_hit


def describe(pop, rows, clone, rev, modes):
    in_pop = set(pop)
    seen = set()
    counts = {"uncovered": 0, "dangling": 0, "bad_pointer": 0, "empty_desc": 0, "dup": 0,
              "dup_desc": 0, "generic_desc": 0, "self_pointer": 0}
    details = []
    blobs, descs = {}, {}
    for row in rows:
        path = row["path"]
        if path not in in_pop:
            counts["dangling"] += 1
            details.append("dangling path=%s" % path)
            continue
        if path in seen:
            counts["dup"] += 1
            details.append("dup path=%s" % path)
            continue
        seen.add(path)
        desc = row["desc"].strip(EMPHASIS)
        if not desc or PLACEHOLDER.match(desc):
            counts["empty_desc"] += 1
            details.append("empty_desc path=%s" % path)
        normalised = norm_desc(desc)
        if normalised in descs and normalised:
            counts["dup_desc"] += 1
            details.append("dup_desc path=%s of=%s" % (path, descs[normalised]))
        else:
            descs.setdefault(normalised, path)
        if path not in blobs:
            try:
                blobs[path] = show(clone, rev, path) if path in modes else b""
            except SetupError:
                blobs[path] = b""
        body_words = {w.group(0).lower() for w in WORD.finditer(
            blobs[path].decode("utf-8", "replace"))}
        shared = desc_words(desc) & body_words
        if len(shared) < MIN_SHARED:
            counts["generic_desc"] += 1
            details.append("generic_desc path=%s shared=%d need=%d" % (
                path, len(shared), MIN_SHARED))
        resolves, self_hit = pointer_state(clone, rev, modes, path, row["pointer"],
                                           in_pop)
        if not resolves:
            counts["bad_pointer"] += 1
            details.append("bad_pointer path=%s pointer=%r" % (
                path, row["pointer"].strip()[:120]))
        if self_hit:
            counts["self_pointer"] += 1
            details.append("self_pointer path=%s pointer=%r" % (
                path, row["pointer"].strip()[:120]))
    missing = sorted(in_pop - seen)
    counts["uncovered"] = len(missing)
    details.extend("uncovered path=%s" % p for p in missing)
    return {
        "population": len(pop),
        "ledger_rows": len(rows),
        "uncovered": counts["uncovered"],
        "dangling": counts["dangling"],
        "bad_pointer": counts["bad_pointer"],
        "empty_desc": counts["empty_desc"],
        "dup": counts["dup"],
        "dup_desc": counts["dup_desc"],
        "generic_desc": counts["generic_desc"],
        "self_pointer": counts["self_pointer"],
    }, details

def checker_selftest(script, work, sandbox):

    text = out + "\n" + err

        return False, "failed", "exit=%s form=%s" % (rc, form)

        return True, "ok", "ratio=%d/%d form=%s" % (passed, total, form)
    if SELFTEST_OK.search(text):
        return True, "ok", "token-ok form=%s" % form
    return False, "failed", "no-ok-marker form=%s" % form


def checker_ledger(script, clone, rev, work, sandbox, mine):

def brace_blocks(text):

    literals are not stripped; the limitation is reported, not hidden, and the pins this counter
    scores carry no brace-bearing comment on a scored line.

        line = raw.strip()
        for char in line:

                kind = classify(header)
                name = function_name(header) if kind == "function" else None
                stack.append({"kind": kind, "header": header, "open": number, "func": name})
            elif char == "}":

                    block = stack.pop()
                    blocks.append((block["kind"], block["header"], block["open"], number))
                    if block["func"]:
                        functions.append((block["func"], block["open"], number))
        if line:

    return functions, blocks

                    "not found\"};` | direct | grounded |")
        rows.append("| P04* | After the scan loop closes, an unresolved keysym returns the error. "
                    "| S370 | pin.cpp:23-24 | `if (!keycode) return {.success = false, .error = "
                    "\"sendshortcut: key not found\"};` | direct | corrected |")
    elif kind == "fabricated":
        rows.append("| P04 | Some claim. | S370 | pin.cpp:15-16 | `this text occurs nowhere in "

def selftest(work):
    pins = os.path.join(work, "selftest-pins")
    os.makedirs(pins, exist_ok=True)
    with open(os.path.join(pins, "pin.cpp"), "w", encoding="utf-8") as handle:
        handle.write(PIN_TEXT)
    cases = [
        ("honest", True, 0, 0),
        ("spliced", False, 1, 0),
        ("misattributed", False, 1, 0),
        ("mislocalized", False, 1, 0),
        ("corrected", True, 0, 0),
        ("superseded", True, 0, 1),
        ("fabricated", False, 1, 0),
        ("no-proposition-column", False, 1, 0),
    ]
    passed = 0
    for kind, want_pass, want_failures, want_superseded in cases:
        result = score_entry(fixture_entry(kind), pins)
        got_pass = result["failures"] == 0 and result["effective"] > 0
        ok = (got_pass == want_pass and result["failures"] == want_failures
              and result["superseded"] == want_superseded)
        passed += 1 if ok else 0
        print("case %-22s want=%s got=%s effective=%d failures=%d superseded=%d" % (
            kind, "PASS" if want_pass else "FAIL", "PASS" if got_pass else "FAIL",
            result["effective"], result["failures"], result["superseded"]))
        for ident, issue in result["problems"][:3]:
            print("  %s: %s" % (ident, issue))
    print("selftest: %d/%d passed" % (passed, len(cases)))
    return 0 if passed == len(cases) else 1


def resolve_work(work):
    created = False
    if work is None:
        work = tempfile.mkdtemp(prefix="c2-r53-", dir=os.environ.get("TMPDIR") or None)
        created = True
    work = os.path.abspath(work)
    try:
        os.makedirs(work, exist_ok=True)
    except OSError as exc:
        raise SetupError("--work %s is not creatable: %s" % (work, exc))
    return work, created


def main(argv=None):
    ap = argparse.ArgumentParser(description="PRED-C2-r53: proposition-bound entry rows")
    ap.add_argument("--repo", default=".")
    ap.add_argument("--rev", default=None)
    ap.add_argument("--entry", default=ENTRY_DEFAULT)
    ap.add_argument("--pins", default=PINS_DEFAULT)
    ap.add_argument("--sources", default=None)
    ap.add_argument("--work", default=None)
    ap.add_argument("--selftest", action="store_true")
    args = ap.parse_args(argv)

    work, created = None, False
    try:
        work, created = resolve_work(args.work)
        if args.selftest:
            return selftest(work)
        if not args.rev:
            raise SetupError("--rev is required unless --selftest")
        repo = os.path.abspath(args.repo)
        rev = resolve_rev(repo, args.rev)
        try:
            text = show(repo, rev, args.entry).decode("utf-8", "replace")
        except SetupError:
            print("C2_R53: props=0 effective=0 superseded=0 failures=1 entry=absent "
                  "verdict=TARGET_MISSED rev=%s" % rev)
            return 1
        pins = args.pins
        if not os.path.isabs(pins):
            pins = os.path.join(repo, pins)
        sources_text = None
        if args.sources:
            try:
                sources_text = show(repo, rev, args.sources).decode("utf-8", "replace")
            except SetupError:
                sources_text = None
        result = score_entry(text, pins, sources_text)
        for ident, issue in result["problems"]:
            print("ROW: id=%s %s" % (ident, issue))
        verdict = "TARGET_MET" if result["failures"] == 0 and result["effective"] > 0 \
            else "TARGET_MISSED"
        print("C2_R53: props=%d effective=%d superseded=%d failures=%d verdict=%s rev=%s" % (
            result["props"], result["effective"], result["superseded"], result["failures"],
            verdict, rev))
        return 0 if verdict == "TARGET_MET" else 1
    except SetupError as exc:
        print("C2_R53: setup=error detail=%s" % exc, file=sys.stderr)
        return 2
    finally:
        if created and work and os.path.isdir(work):
            shutil.rmtree(work, ignore_errors=True)


if __name__ == "__main__":
    sys.exit(main())
