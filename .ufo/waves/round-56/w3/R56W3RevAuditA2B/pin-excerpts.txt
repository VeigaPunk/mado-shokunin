80: void CConfigWatcher::onInotifyEvent() {
81:     constexpr size_t                                     BUFFER_SIZE = sizeof(inotify_event) + NAME_MAX + 1;
82:     alignas(inotify_event) std::array<char, BUFFER_SIZE> buffer      = {};
83:     const ssize_t                                        bytesRead   = read(m_inotifyFd.get(), buffer.data(), buffer.size());
84:     if (bytesRead <= 0)
85:         return;
86: 
87:     for (size_t offset = 0; offset < sc<size_t>(bytesRead);) {
88:         const auto* ev = rc<const inotify_event*>(buffer.data() + offset);
89: 
90:         if (offset + sizeof(inotify_event) > sc<size_t>(bytesRead)) {
91:             Log::logger->log(Log::ERR, "CConfigWatcher: malformed inotify event, truncated header");
92:             break;
93:         }
94: 
95:         if (offset + sizeof(inotify_event) + ev->len > sc<size_t>(bytesRead)) {
96:             Log::logger->log(Log::ERR, "CConfigWatcher: malformed inotify event, truncated name field");
97:             break;
98:         }
99: 
100:         const auto WD = std::ranges::find_if(m_watches, [wd = ev->wd](const auto& e) { return e.wd == wd; });
101: 
102:         if (WD == m_watches.end())
103:             Log::logger->log(Log::ERR, "CConfigWatcher: got an event for wd {} which we don't have?!", ev->wd);
104:         else
105:             m_watchCallback(SConfigWatchEvent{
106:                 .file = WD->file,
107:             });
108: 
109:         offset += sizeof(inotify_event) + ev->len;
110:     }
111: }
