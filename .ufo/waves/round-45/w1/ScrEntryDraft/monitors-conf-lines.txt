6:# env = GDK_SCALE,2
7:# monitor=,preferred,auto,auto

12:# Good compromise for 27" or 32" 4K monitors (but fractional!)
13:# env = GDK_SCALE,1.75

16:# Straight 1x setup for low-resolution displays like 1080p or 1440p
17:# Or for ultrawide monitors like 34" 3440x1440 or 49" 5120x1440
18:# env = GDK_SCALE,1
19:# monitor=,preferred,auto,1
20:
21:# Portrait/rotated secondary monitor (transform: 1 = 90°, 3 = 270°)
