--- stof.cpp
#include <cstdio>
#include <string>
#include <exception>
int main() {
    const char* v[] = {"1e-46", "-1e-46", "1e-45", "1e-40", "1.17549428e-38", "1.17549430e-38",
                       "1.1754942e-38", "1e39", "-1e39", "0x10", "0x1p3", "0.7x", "nan", "inf", "fast", "0.335", "5.", ".5", ""};
    for (auto s : v) {
        try {
            float f = std::stof(std::string{s});
            printf("%-14s ok float=%.9g bits_ok\n", s, (double)f);
        } catch (const std::exception& e) {
            printf("%-14s THROW %s\n", s, e.what());
        }
    }
    return 0;
}
