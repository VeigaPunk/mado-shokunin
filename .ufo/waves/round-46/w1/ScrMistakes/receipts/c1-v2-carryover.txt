def make_fakebin(d, trip_fd):
    os.makedirs(d)
    for n in ("hyprctl", "libinput", "journalctl", "udevadm"):
        p = os.path.join(d, n)
        with open(p, "w") as f:
            f.write("#!/bin/sh\nprintf x >&%d\nexit 0\n" % trip_fd)
        os.chmod(p, 0o755)

def root_links():
    out = []
    for n in ("bin", "sbin", "lib", "lib64"):
        p = "/" + n
        if os.path.islink(p):
            out += ["--symlink", os.readlink(p), p]
    return out
